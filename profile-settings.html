-- 1. Enable Row Level Security on profiles table alter table public.profiles enable row level security; -- 2. Allow anyone (anon & authenticated) to read public profile info create policy "Public profiles are viewable by everyone" on public.profiles for select using (true); -- 3. Allow authenticated users to insert their own profile row create policy "Users can insert their own profile" on public.profiles for insert to authenticated with check (auth.uid() = id); -- 4. Allow authenticated users to update their own profile row create policy "Users can update their own profile" on public.profiles for update to authenticated using (auth.uid() = id) with check (auth.uid() = id);
