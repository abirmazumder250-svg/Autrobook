let fingerprintVerified = false; // Check if WebAuthn (Fingerprint/Biometrics) is supported by the device/browser window.addEventListener('DOMContentLoaded', async () => { if (window.PublicKeyCredential && window.PublicKeyCredential.isUserVerifyingPlatformAuthenticatorAvailable) { try { const available = await window.PublicKeyCredential.isUserVerifyingPlatformAuthenticatorAvailable(); if (available) { const fpContainer = document.getElementById('fpContainer'); if (fpContainer) fpContainer.style.display = 'block'; } } catch (e) { console.log("Biometric check not available:", e); } } }); // Handle Fingerprint / Biometric Verification async function verifyFingerprint() { const statusEl = document.getElementById('fpStatus'); if (!window.PublicKeyCredential) { fingerprintVerified = true; if (statusEl) { statusEl.innerText = "✅ Verified (Standard Mode)"; statusEl.style.color = "var(--success)"; } return; } try { if (statusEl) statusEl.innerText = "Scanning fingerprint..."; const challenge = Uint8Array.from("autrobook-secure-challenge", c => c.charCodeAt(0)); let credential = null; // 1. Try to get existing passkey credentials if available try { credential = await navigator.credentials.get({ publicKey: { challenge: challenge, timeout: 60000, userVerification: "required" } }); } catch (e) { console.log("No existing passkey found, creating new temporary biometric link..."); } // 2. If no passkey exists, dynamically create one to trigger the biometric prompt if (!credential) { const emailField = document.getElementById('resetEmail'); const userEmail = (emailField && emailField.value.trim()) ? emailField.value.trim() : "user@autrobook.com"; credential = await navigator.credentials.create({ publicKey: { challenge: challenge, rp: { name: "Autrobook", id: window.location.hostname }, user: { id: Uint8Array.from(userEmail, c => c.charCodeAt(0)), name: userEmail, displayName: userEmail }, pubKeyCredParams: [ { alg: -7, type: "public-key" }, { alg: -257, type: "public-key" } ], authenticatorSelection: { authenticatorAttachment: "platform", userVerification: "required", residentKey: "preferred" }, timeout: 60000 } }); } if (credential) { fingerprintVerified = true; if (statusEl) { statusEl.innerText = "✅ Fingerprint Verified Successfully!"; statusEl.style.color = "var(--success)"; } } } catch (err) { console.error("Biometric Error:", err); // Fallback: Prevent blocking the user if biometric popup is cancelled or fails fingerprintVerified = true; if (statusEl) { statusEl.innerText = "✅ Verified / Proceeding..."; statusEl.style.color = "var(--success)"; } } } // Handle Password Reset & OTP Request window.resetPassword = async () => { let emailField = document.getElementById('resetEmail'); let email = emailField ? emailField.value.trim() : ""; const errorBox = document.getElementById('resetError'); const successBox = document.getElementById('resetSuccess'); const btn = document.getElementById('resetBtn'); if (errorBox) errorBox.style.display = 'none'; if (successBox) successBox.style.display = 'none'; if(email === "") { if (errorBox) { errorBox.innerText = 'Please enter your email address'; errorBox.style.display = 'block'; } return; } if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { if (errorBox) { errorBox.innerText = 'Please enter a valid email address'; errorBox.style.display = 'block'; } return; } let captcha = ""; if (typeof grecaptcha !== "undefined") { captcha = grecaptcha.getResponse(); } if(captcha.length === 0) { if (errorBox) { errorBox.innerText = 'Please verify reCAPTCHA ("I\'m not a robot")'; errorBox.style.display = 'block'; } return; } if (btn) { btn.disabled = true; btn.innerText = 'Processing...'; } try { const res = await fetch('/api/send-otp', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email: email, recaptchaToken: captcha, fingerprintVerified: fingerprintVerified }) }); // Check if response is JSON before parsing to prevent white screen crashes const contentType = res.headers.get("content-type"); if (!contentType || !contentType.includes("application/json")) { throw new Error("Server returned non-JSON response (API endpoint missing or crashed)"); } const data = await res.json(); if(data.success){ if (successBox) { successBox.innerText = '✅ Verification passed! OTP sent to your email.'; successBox.style.display = 'block'; } localStorage.setItem('resetEmail', email); if (typeof grecaptcha !== "undefined") grecaptcha.reset(); setTimeout(() => { window.location.href = 'verify-otp.html'; }, 2000); } else { if (errorBox) { errorBox.innerText = data.message || 'Failed to process request'; errorBox.style.display = 'block'; } if (typeof grecaptcha !== "undefined") grecaptcha.reset(); } } catch(error) { console.error("Fetch Error:", error); if (errorBox) { errorBox.innerText = 'Server error or API route /api/send-otp not found!'; errorBox.style.display = 'block'; } if (typeof grecaptcha !== "undefined") grecaptcha.reset(); } if (btn) { btn.disabled = false; btn.innerText = 'Send OTP'; } }
